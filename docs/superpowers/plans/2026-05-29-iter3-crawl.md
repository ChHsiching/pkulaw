def export_csv(cases: list[dict], path: Path) -> None:
    """Export cases to CSV with UTF-8 BOM, dynamic columns, sanitized values."""
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)

    if not cases:
        path.write_text("", encoding="utf-8")
        return

    columns = _build_column_order(cases)

    with open(path, "w", encoding="utf-8-sig", newline="") as f:
        writer = csv.writer(f)
        writer.writerow(columns)
        for case in cases:
            row = []
            for col in columns:
                val = _sanitize(case.get(col, ""))
                if isinstance(val, str):
                    val = val.replace("\r\n", " ").replace("\n", " ").replace("\r", " ")
                row.append(val)
            writer.writerow(row)

```python
"""Crawler for PKULaw cases — partitioning-based search + robust fetch."""

import json
import time
from datetime import datetime
from pathlib import Path

from src.auth import (
    authenticate,
    close_browser,
    launch_browser,
    reauthenticate,
    search_api,
)
from src.exporter import export_excel, export_json
from src.parser import parse_case
from src.partition import SORT_ORDERS, _add_dimension_filter, partition_query
from src.query import PAGE_SIZE, build_api_body

INTERMEDIATE_INTERVAL = 500


def _progress_file(output_dir: Path) -> Path:
    return output_dir / "progress.json"


def _search_cache_file(output_dir: Path) -> Path:
    return output_dir / "search_results.json"


def _results_file(output_dir: Path) -> Path:
    return output_dir / "pkulaw_cases.json"


def _excel_file(output_dir: Path) -> Path:
    return output_dir / "pkulaw_cases.xlsx"


def _load_progress(output_dir: Path) -> set[str]:
    path = _progress_file(output_dir)
    if path.exists():
        data = json.loads(path.read_text())
        return set(data.get("fetched_gids", []))
    return set()


def _save_progress(output_dir: Path, fetched_gids: set[str]) -> None:
    output_dir.mkdir(parents=True, exist_ok=True)
    _progress_file(output_dir).write_text(
        json.dumps({"fetched_gids": sorted(fetched_gids)}, ensure_ascii=False, indent=2)
    )


def _matches_query(cached: dict, search_config: dict) -> bool:
    """Check if cached search_results.json matches the current query."""
    if "query" not in cached:
        return False
    cached_nodes = cached["query"].get("fieldNodes", [])
    current_nodes = search_config.get("fieldNodes", [])
    return cached_nodes == current_nodes


def _collect_leaf_searches(tree: dict, base_config: dict) -> list[tuple[str, dict]]:
    """Flatten partition tree to list of (label, search_config) for leaf nodes."""
    if tree.get("children") is None:
        return [(tree.get("label", ""), base_config)]
    results = []
    dim = tree["dimension"]
    for child in tree["children"]:
        child_config = _add_dimension_filter(base_config, dim, child["label"])
        results.extend(_collect_leaf_searches(child, child_config))
    return results


def _save_search_results(
    output_dir: Path,
    results: list[dict],
    search_config: dict,
    partition_tree: dict,
    started_at: str,
    completed_at: str,
) -> None:
    """Save enhanced search_results.json with query and partition metadata."""
    output_dir.mkdir(parents=True, exist_ok=True)

    dimensions = []
    node = partition_tree
    while node.get("children") is not None:
        dimensions.append(node["dimension"])
        node = node["children"][0] if node["children"] else node

    data = {
        "query": {
            "fieldNodes": search_config.get("fieldNodes", []),
            "started_at": started_at,
            "completed_at": completed_at,
        },
        "partition_strategy": {
            "dimensions": dimensions,
            "sort_orders": SORT_ORDERS,
            "total_groups": partition_tree.get("groups", 0),
        },
        "total_unique": len(results),
        "results": results,
    }
    _search_cache_file(output_dir).write_text(
        json.dumps(data, ensure_ascii=False, indent=2)
    )


def run_search(
    search_config: dict,
    page,
    token: str,
    output_dir: Path,
    logger,
) -> list[dict]:
    """Search with recursive partitioning. Returns list of {gid, title}."""
    output_dir.mkdir(parents=True, exist_ok=True)
    cache_path = _search_cache_file(output_dir)

    if cache_path.exists():
        cached = json.loads(cache_path.read_text())
        if _matches_query(cached, search_config):
            logger.info("Search cache matches query, skipping search")
            results = cached.get("results", cached) if isinstance(cached, dict) else cached
            print(f"搜索缓存匹配，跳过搜索 ({len(results)} 条)")
            return results

    max_pages = search_config.get("settings", {}).get("max_pages", 10)
    delay = search_config.get("settings", {}).get("delay", 0.3)

    print("\n=== 搜索阶段 ===")
    logger.info("Starting search phase with recursive partitioning")
    started_at = datetime.now().isoformat()

    partition_tree = partition_query(page, token, search_config, max_pages=max_pages)
    logger.info(
        "Partition complete: %d total, %d crawlable, %d groups",
        partition_tree["total"],
        partition_tree["crawlable"],
        partition_tree["groups"],
    )

    leaves = _collect_leaf_searches(partition_tree, search_config)
    print(f"分区策略：{len(leaves)} 个分区，预估 {partition_tree['crawlable']:,} 条")

    results: list[dict] = []
    seen_gids: set[str] = set()

    for leaf_idx, (label, leaf_config) in enumerate(leaves):
        leaf_label = label if label else f"分区 {leaf_idx + 1}"
        print(f"  搜索 {leaf_idx + 1}/{len(leaves)}: {leaf_label}")
        logger.info("Searching partition %d/%d: %s", leaf_idx + 1, len(leaves), leaf_label)

        for sort_order in SORT_ORDERS:
            for page_idx in range(max_pages):
                body = build_api_body(leaf_config, page_idx, PAGE_SIZE, sort_order)
                data = search_api(page, token, body)
                items = data.get("data", [])

                if not items:
                    break

                new_count = 0
                for item in items:
                    gid = item["gid"]
                    if gid in seen_gids:
                        continue
                    seen_gids.add(gid)
                    results.append({"gid": gid, "title": item.get("title", "")})
                    new_count += 1

                if page_idx == 0:
                    total = data.get("total", 0)
                    logger.info("  %s [%s]: total=%d, +%d new", leaf_label, sort_order, total, new_count)

                if new_count == 0:
                    break

                time.sleep(delay)

        completed_at = datetime.now().isoformat()
        _save_search_results(output_dir, results, search_config, partition_tree, started_at, completed_at)
        print(f"    累计：{len(results)} 条")

    completed_at = datetime.now().isoformat()
    _save_search_results(output_dir, results, search_config, partition_tree, started_at, completed_at)

    print(f"\n搜索完成：{len(results)} 条唯一结果")
    logger.info("Search complete: %d unique results", len(results))
    return results


def _save_all(
    results: list[dict], fetched_gids: set[str], output_dir: Path, logger
) -> None:
    export_json(results, _results_file(output_dir))
    try:
        export_excel(results, _excel_file(output_dir))
    except Exception as e:
        logger.warning("Excel export failed: %s", e)
    _save_progress(output_dir, fetched_gids)
    good = sum(1 for c in results if len(c.get("full_text", "")) > 500)
    print(f"  -> Saved ({len(results)} cases, {len(fetched_gids)} fetched, {good} loaded)")
    logger.info("Saved: %d cases, %d fetched, %d loaded", len(results), len(fetched_gids), good)


def run_fetch(
    search_config: dict,
    output_dir: Path,
    logger,
) -> list[dict]:
    """Fetch detail pages for cached search results. Auto-restarts on failure."""
    output_dir.mkdir(parents=True, exist_ok=True)

    cache_path = _search_cache_file(output_dir)
    if not cache_path.exists():
        print("No search cache. Run search phase first.")
        return []

    cache_data = json.loads(cache_path.read_text())
    cases_meta = (
        cache_data.get("results", []) if isinstance(cache_data, dict) else cache_data
    )

    delay = search_config.get("settings", {}).get("delay", 0.3)
    browser_path = search_config.get("settings", {}).get("browser", "/usr/bin/chromium")
    headless = search_config.get("settings", {}).get("headless", True)

    while True:
        fetched_gids = _load_progress(output_dir)
        results: list[dict] = []
        if _results_file(output_dir).exists():
            results = json.loads(_results_file(output_dir).read_text())

        remaining = [c for c in cases_meta if c["gid"] not in fetched_gids]
        print(f"\n{'='*50}")
        print(f"Remaining: {len(remaining)}/{len(cases_meta)} ({len(fetched_gids)} fetched)")
        logger.info("Fetch phase: %d remaining", len(remaining))

        if not remaining:
            print("All fetched!")
            _save_all(results, fetched_gids, output_dir, logger)
            return results

        try:
            pw, browser, context, page = launch_browser(
                browser_path=browser_path, headless=headless,
            )
            token = authenticate(page)
            print(f"=== Fetching {len(remaining)} cases ===")

            consecutive_errors = 0
            for i, case in enumerate(remaining):
                gid = case["gid"]
                title = case["title"]
                url = f"https://www.pkulaw.com/pfnl/{gid}.html"

                try:
                    for nav_attempt in range(3):
                        try:
                            page.goto(url, wait_until="commit", timeout=60000)
                            break
                        except Exception:
                            if nav_attempt == 2:
                                raise
                            page.wait_for_timeout(5000)

                    for _ in range(16):
                        page.wait_for_timeout(500)
                        html_len = page.evaluate(
                            "() => document.querySelector('.fulltext-wrap')?.innerHTML?.length || 0"
                        )
                        if html_len > 1000:
                            break

                    html = page.content()
                    parsed = parse_case(html, gid)
                    if not parsed["title"] or parsed["title"] in ("已进入法宝V6", ""):
                        parsed["title"] = title

                    results.append(parsed)
                    fetched_gids.add(gid)
                    consecutive_errors = 0

                    if (i + 1) % 50 == 0:
                        print(f"[{i+1}/{len(remaining)}] ({title[:40]})")

                except Exception as e:
                    err_str = str(e)[:80]
                    consecutive_errors += 1
                    print(f"[{i+1}/{len(remaining)}] ERR: {err_str}")
                    logger.error("[%d/%d] %s: %s", i + 1, len(remaining), title[:30], err_str)

                    if "Execution context" in err_str or "Target closed" in err_str:
                        try:
                            page = context.new_page()
                            token = reauthenticate(page)
                        except Exception:
                            print("  Page recovery failed, restarting browser...")
                            logger.warning("Page recovery failed, restarting browser")
                            break

                    if consecutive_errors >= 5:
                        print("  Too many errors, restarting browser...")
                        logger.warning("5 consecutive errors, restarting browser")
                        break

                if len(fetched_gids) % INTERMEDIATE_INTERVAL == 0:
                    _save_all(results, fetched_gids, output_dir, logger)

                time.sleep(delay)

            close_browser(pw, browser)

        except Exception as e:
            print(f"Session crashed: {e}")
            logger.error("Session crashed: %s", e)

        _save_all(results, fetched_gids, output_dir, logger)
        print("Restart in 15s...")
        time.sleep(15)

    good = sum(1 for c in results if len(c.get("full_text", "")) > 500)
    print(f"\nTotal: {len(results)} cases, {good} loaded ({100*good//max(len(results),1)}%)")
    logger.info("Fetch complete: %d cases, %d loaded", len(results), good)
    return results

def cmd_crawl(config: dict) -> None:
    """Execute full crawl: search + fetch + export."""
    from pathlib import Path

    from src.auth import authenticate, close_browser, launch_browser
    from src.crawler import run_fetch, run_search
    from src.exporter import export_csv, export_excel, export_json

    output_dir = Path(config["settings"]["output_dir"])
    logger = setup_logger(output_dir / "crawl.log")

    logger.info("Crawl command started")
    logger.info("Query: %s", _format_query(config))

    print("\n=== PKULaw 采集 ===")
    print("\n检索条件：")
    for node in config["fieldNodes"]:
        field = node["field"]
        if "value" in node:
            print(f"  {field}: {node['value']}")
        elif "values" in node:
            print(f"  {field}: {', '.join(node['values'])}")
        elif "range" in node:
            print(f"  {field}: {node['range'][0]} ~ {node['range'][1]}")

    # Phase 1: Search
    print("\n--- 阶段 1/3: 搜索 ---")
    pw, browser, context, page = launch_browser(
        browser_path=config["settings"].get("browser", "/usr/bin/chromium"),
        headless=config["settings"].get("headless", True),
    )
    try:
        token = authenticate(page)
        logger.info("Authenticated (%s...)", token[:30])
        print(f"认证成功 ({token[:30]}...)")
        search_results = run_search(config, page, token, output_dir, logger)
    finally:
        close_browser(pw, browser)

    # Phase 2: Fetch
    print("\n--- 阶段 2/3: 采集 ---")
    cases = run_fetch(config, output_dir, logger)

    # Phase 3: Export
    print("\n--- 阶段 3/3: 导出 ---")
    formats = config["settings"].get("format", ["json"])
    if "json" in formats:
        export_json(cases, output_dir / "pkulaw_cases.json")
        print(f"  JSON: {output_dir / 'pkulaw_cases.json'}")
    if "xlsx" in formats:
        export_excel(cases, output_dir / "pkulaw_cases.xlsx")
        print(f"  Excel: {output_dir / 'pkulaw_cases.xlsx'}")
    if "csv" in formats:
        export_csv(cases, output_dir / "pkulaw_cases.csv")
        print(f"  CSV: {output_dir / 'pkulaw_cases.csv'}")

    good = sum(1 for c in cases if len(c.get("full_text", "")) > 500)
    print(f"\n采集完成：{len(cases)} 条 ({good} 条有效)")
    logger.info("Crawl complete: %d cases, %d valid", len(cases), good)
```

Add imports at the top of `pkulaw.py`. The `cmd_crawl` function uses local imports (inside the function body) for `run_search`, `run_fetch`, `export_csv`, etc. — same pattern as `cmd_estimate`. The `export_csv` import is needed; add it alongside the existing local imports.
